import pathlib, subprocess, hashlib, json
r=pathlib.Path.cwd()
paths=subprocess.check_output(['git','ls-files','-co','--exclude-standard','-z']).decode().split('\0')
print(json.dumps({p:hashlib.sha256((r/p).read_bytes()).hexdigest() for p in sorted(set(paths)) if p and (r/p).is_file() and not p.startswith('evidence/g7/ipc-parent-regression/')},sort_keys=True,indent=2))
