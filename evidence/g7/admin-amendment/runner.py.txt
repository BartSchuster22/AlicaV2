#!/usr/bin/env python3
"""Exact allowlist development transfer/logging; never stages or deletes remote files."""
import sys, subprocess, pathlib, json, hashlib, tarfile, io, datetime
ROOT=pathlib.Path('/home/herman/g7-implementation')
E=ROOT/'evidence/g7/admin-amendment'
SSH=['ssh','-i','/home/herman/.ssh/alica_v2_vps_ed25519','-o','IdentitiesOnly=yes','-o','BatchMode=yes','-o','StrictHostKeyChecking=yes','-o','UserKnownHostsFile=/home/herman/.ssh/alica_v2_known_hosts','alica-dev@167.233.135.142']
REMOTE='/home/alica-dev/AlicaV2'
def run(label, command, data=None):
    assert not (E/(label+'.log')).exists(), label
    with (E/(label+'.log')).open('wb') as f:
        p=subprocess.run(SSH+[command],input=data,stdout=f,stderr=subprocess.STDOUT)
    (E/(label+'.exit')).write_text(str(p.returncode)+'\n')
    with (E/'commands.jsonl').open('a') as f: f.write(json.dumps({'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'label':label,'command':command,'exit':p.returncode})+'\n')
    print(label,p.returncode,flush=True)
    return p.returncode
probe="""import subprocess,pathlib,hashlib,json
r=pathlib.Path('.'); names=set(subprocess.check_output(['git','ls-files','-z']).decode().split('\\0')+subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z']).decode().split('\\0')); names.discard('')
print(json.dumps({'head':subprocess.check_output(['git','rev-parse','HEAD']).decode().strip(),'status':subprocess.check_output(['git','status','--short']).decode(),'files':{n:hashlib.sha256((r/n).read_bytes()).hexdigest() for n in sorted(names) if (r/n).is_file()}},indent=2))
"""
mode,label=sys.argv[1:3]
if mode=='probe':
    run(label,'cd '+REMOTE+' && python3 -c '+__import__('shlex').quote(probe))
elif mode=='run':
    sys.exit(run(label,'cd '+REMOTE+' && export PATH="$PWD/.tools/node/bin:$PATH"; '+sys.argv[3]))
elif mode=='send':
    names=(E/'transfer-allowlist.txt').read_text().splitlines()
    if len(sys.argv)>3:
        selected=(E/sys.argv[3]).read_text().splitlines()
        assert set(selected)<=set(names)
        names=selected
    assert len(names)==len(set(names)) and all(n and not n.startswith('/') and '..' not in pathlib.PurePosixPath(n).parts for n in names)
    b=io.BytesIO()
    with tarfile.open(fileobj=b,mode='w') as t:
        for n in names:t.add(ROOT/n,arcname=n,recursive=False)
    (E/(label+'-input.tar')).write_bytes(b.getvalue())
    (E/(label+'-input.json')).write_text(json.dumps({n:hashlib.sha256((ROOT/n).read_bytes()).hexdigest() for n in names},indent=2)+'\n')
    sys.exit(run(label,'cd '+REMOTE+' && tar -xf -',b.getvalue()))
elif mode=='return':
    names=(E/'transfer-allowlist.txt').read_text().splitlines()
    b=subprocess.check_output(SSH+['cd '+REMOTE+' && tar -cf - -- '+ ' '.join(__import__('shlex').quote(n) for n in names)])
    (E/(label+'.tar')).write_bytes(b)
    with tarfile.open(fileobj=io.BytesIO(b)) as t:
        assert set(t.getnames())==set(names)
        for n in names:
            assert t.getmember(n).isfile()
            (ROOT/n).write_bytes(t.extractfile(n).read())
    (E/(label+'-hashes.json')).write_text(json.dumps({n:hashlib.sha256((ROOT/n).read_bytes()).hexdigest() for n in names},indent=2)+'\n')
else: raise ValueError(mode)
