#!/usr/bin/env python3
"""Return only this checkpoint's process evidence, preserving archive and bytes."""
import pathlib,subprocess,json,tarfile,io,hashlib
E=pathlib.Path('/home/herman/g7-implementation/evidence/g7/admin-amendment')
ssh=['ssh','-i','/home/herman/.ssh/alica_v2_vps_ed25519','-o','IdentitiesOnly=yes','-o','BatchMode=yes','-o','StrictHostKeyChecking=yes','-o','UserKnownHostsFile=/home/herman/.ssh/alica_v2_known_hosts','alica-dev@167.233.135.142']
command='cd /home/alica-dev/AlicaV2 && tar -czf - -- evidence/g7/admin-amendment-process'
archive=E/'process-evidence.tar.gz'
assert not archive.exists()
with archive.open('xb') as out,(E/'process-transfer.stderr.log').open('xb') as err:
    result=subprocess.run(ssh+[command],stdout=out,stderr=err)
(E/'process-transfer.exit').write_text(str(result.returncode)+'\n')
assert result.returncode==0
expected=json.loads((E/'after-full.log').read_text())['files']
files={}
with tarfile.open(archive,'r:gz') as tar:
    for member in tar.getmembers():
        if member.isdir():continue
        assert member.isfile(),member.name
        prefix='evidence/g7/admin-amendment-process/'
        assert member.name.startswith(prefix)
        relative=pathlib.PurePosixPath(member.name[len(prefix):])
        assert not relative.is_absolute() and '..' not in relative.parts
        data=tar.extractfile(member).read()
        digest=hashlib.sha256(data).hexdigest()
        assert digest==expected[member.name],member.name
        dest=E/'process'/relative
        dest.parent.mkdir(parents=True,exist_ok=True)
        with dest.open('xb') as f:f.write(data)
        files[member.name]={'local':str(dest.relative_to(E)), 'sha256':digest}
assert set(files)=={n for n in expected if n.startswith('evidence/g7/admin-amendment-process/')}
receipt={'command':command,'archiveSHA256':hashlib.sha256(archive.read_bytes()).hexdigest(),'archiveBytes':archive.stat().st_size,'files':files}
(E/'process-transfer-receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print('Verified returned process evidence files:',len(files),'archive bytes:',receipt['archiveBytes'])
